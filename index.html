-- Shared Portfolio Plan table create table if not exists public.portfolio_plan ( id bigint generated always as identity primary key, sector text not null, ticker text not null unique, company_name text, target_weight numeric, updated_at timestamptz default now() ); alter table public.portfolio_plan enable row level security; grant select, insert, update, delete on table public.portfolio_plan to authenticated; grant usage, select on sequence public.portfolio_plan_id_seq to authenticated; drop policy if exists "Authenticated users can view portfolio plan" on public.portfolio_plan; create policy "Authenticated users can view portfolio plan" on public.portfolio_plan for select to authenticated using (true); drop policy if exists "Instructor can insert portfolio plan" on public.portfolio_plan; create policy "Instructor can insert portfolio plan" on public.portfolio_plan for insert to authenticated with check (public.current_user_role() = 'Instructor'); drop policy if exists "Instructor can update portfolio plan" on public.portfolio_plan; create policy "Instructor can update portfolio plan" on public.portfolio_plan for update to authenticated using (public.current_user_role() = 'Instructor') with check (public.current_user_role() = 'Instructor'); drop policy if exists "Instructor can delete portfolio plan" on public.portfolio_plan; create policy "Instructor can delete portfolio plan" on public.portfolio_plan for delete to authenticated using (public.current_user_role() = 'Instructor');
